Imperative Stream Boilerplate vs. ShreePdf Fluent Builder

iText 7 (Verbose Stream & Cell Setup) Legacy API
using iText.Kernel.Pdf; using iText.Layout; using iText.Layout.Element; using var writer = new PdfWriter("statement.pdf"); using var pdf = new PdfDocument(writer); using var document = new Document(pdf); document.Add(new Paragraph("Account Statement").SetBold().SetFontSize(18)); Table table = new Table(new float[] { 200, 100 }); table.AddHeaderCell("Transaction"); table.AddHeaderCell("Amount"); table.AddCell("Cloud Compute"); table.AddCell("$450.00"); document.Add(table);
ShreePdf (Modern Theme + Auto-Fit) Clean C# 12
using ShreePdf.Fluent; Document.Create("Account Statement") .WithTheme(ThemePresets.CorporateBlue) .AddHeading("Account Statement", level: 1) .AddTable(t => { t.Columns.Add(new TableColumn { Header = "Transaction", WidthRatio = 2 }); t.Columns.Add(new TableColumn { Header = "Amount", WidthRatio = 1 }); t.Rows.Add(new TableRow { Cells = { "Cloud Compute", "$450.00" } }); }) .Save("statement.pdf");
